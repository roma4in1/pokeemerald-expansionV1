#ifndef GUARD_CONSTANTS_REGION_MAP_SECTIONS_H
#define GUARD_CONSTANTS_REGION_MAP_SECTIONS_H

## for map_section in map_sections
    {{ map_section.id }},
## endfor
    MAPSEC_NONE,
    MAPSEC_COUNT
};

// Special location IDs that use the same value space as MAPSECs.
// PELAGIOS (2026-06-14): the Pokemon metLocation field is a 9-bit bitfield now
// (see struct PokemonSubstruct3 in include/pokemon.h) so MAPSEC ids may grow to
// 508 for Primalis and later islands. These sentinels moved from 0xFD/0xFE/0xFF
// to the TOP of the 9-bit range (509/510/511) so a real MAPSEC id can never
// collide with them as the enum grows past the old u8 ceiling. The runtime
// handler type (mapsec_u8_t / metloc_u8_t, gametypes.h) is u16, which holds these.
#define METLOC_SPECIAL_EGG         0x1FD
#define METLOC_IN_GAME_TRADE       0x1FE
#define METLOC_FATEFUL_ENCOUNTER   0x1FF

#define KANTO_MAPSEC_START  MAPSEC_PALLET_TOWN
#define KANTO_MAPSEC_END    MAPSEC_SPECIAL_AREA
#define KANTO_MAPSEC_COUNT (KANTO_MAPSEC_END - KANTO_MAPSEC_START + 1)

#endif // GUARD_CONSTANTS_REGION_MAP_SECTIONS_H
